cmake_minimum_required(VERSION 3.28)
project(sm64_ar C CXX)
if(APPLE)
  enable_language(OBJC OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(cmake/Toolchain.cmake)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
option(SM64AR_TESTS "Build test-only seams (no tests ship in this tree)" OFF)
option(SM64AR_SM64 "Build the SM64 game adapter (games/sm64)" ON)
if(NOT APPLE AND NOT ANDROID)
  option(SM64AR_HOST_TESTS "Build the portable ctest suite" ON)
else()
  option(SM64AR_HOST_TESTS "Build the portable ctest suite" OFF)
endif()
option(SM64AR_QUEST "Build the Meta Quest app" OFF)
if(ANDROID)
  set(SM64AR_QUEST ON)
endif()
# The Quest app as a Linux desktop OpenXR program, run on Monado's simulated headset (scripts/desktop_xr.py).
option(SM64AR_DESKTOP_XR "Build the Quest app for desktop OpenXR (Linux, Monado)" OFF)
if(SM64AR_DESKTOP_XR AND (APPLE OR ANDROID))
  message(FATAL_ERROR "SM64AR_DESKTOP_XR is for Linux desktop builds")
endif()
enable_testing()
if(SM64AR_QUEST)
  # No host path in the binary (__FILE__, debug info); C and C++, every directory below.
  add_compile_options(-ffile-prefix-map=${PROJECT_SOURCE_DIR}=.)
endif()
add_subdirectory(runtime)
if(SM64AR_QUEST OR SM64AR_HOST_TESTS OR SM64AR_DESKTOP_XR)
  add_subdirectory(apps/quest/panel)   # the settings panel's pure parts (the Quest app and the host tests)
  # The pure half of the APK-asset copy (manifest parsing, the up-to-date check): the Quest app and the host tests.
  add_library(quest_bundled STATIC apps/quest/bundled_copy_pure.cpp)
  target_include_directories(quest_bundled PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/apps/quest)
endif()
if(SM64AR_SM64)
  add_subdirectory(games/sm64)
endif()
if(SM64AR_VISIONOS)
  # An ImmersiveSpace can only be declared from a SwiftUI App, and CMake builds Swift only with the Xcode generator.
  enable_language(Swift)
  add_subdirectory(apps/visionos)
elseif(SM64AR_QUEST)
  add_subdirectory(apps/quest)
elseif(SM64AR_DESKTOP_XR)
  add_subdirectory(apps/desktop_xr)
elseif(APPLE)
  add_subdirectory(apps/macos_harness)
endif()
if(SM64AR_HOST_TESTS)
  add_subdirectory(runtime/tests)
endif()
if(NOT APPLE AND NOT ANDROID AND SM64AR_SM64)
  add_subdirectory(apps/pack_tool)
endif()

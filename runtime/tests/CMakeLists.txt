  target_link_libraries(${name} PRIVATE ar_core ar_platform_fake ar_cube_adapter ar_render_common ar_vos_logic ar_qx_convert)

  set_tests_properties(${name} PROPERTIES SKIP_RETURN_CODE 77)
endfunction()
